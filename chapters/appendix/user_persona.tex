\newcommand{\upNombre}{Valeria G.}
\newcommand{\upEdad}{47 años}
\newcommand{\upOcupacion}{Docente de escuela primaria}
\newcommand{\upUbicacion}{Buenos Aires, Argentina}
\newcommand{\upRol}{Familiar cuidadora a distancia}
\newcommand{\upArquetipo}{Mamá de dos hijos y un adulto a cargo}

\newcommand{\upContexto}{Su madre de 76 años vive sola en otro barrio con deterioro
cognitivo leve: está activa pero olvida cosas y no maneja tecnología.
Valeria trabaja como docente, cría a dos hijos y comparte el cuidado
con sus hermanos visitándola día por medio.}

\newcommand{\upCita}{``Necesitaría algo que me haga sentir más cerca de ella, sin que ella tenga que hacer nada.''}

\newcommand{\upObjetivos}{%
  \upItem{upGreen}{Saber que su madre está bien sin llamarla constantemente.}
  \upItem{upGreen}{Recibir alerta ante caída o inactividad prolongada.}
  \upItem{upGreen}{No requerir que el adulto mayor use dispositivos.}
  \upItem{upGreen}{Coordinarse con sus hermanos con info objetiva.}}

\newcommand{\upFrustraciones}{%
  \upItem{upRed}{El celular es el único canal y no siempre es atendido.}
  \upItem{upRed}{No existen soluciones accesibles y fáciles de instalar.}
  \upItem{upRed}{Siente culpa e incertidumbre cuando no puede estar presente.}
  \upItem{upRed}{Le genera dudas instalar cámaras por la privacidad.}}

\newcommand{\upComportamientos}{%
  \upItem{upAccent}{Llama a su madre a la mañana y a la tarde-noche, todos los días.}
  \upItem{upAccent}{Coordina visitas con sus hermanos para cubrir toda la semana.}
  \upItem{upAccent}{Piensa en momentos clave: si se levantó, si cocinó, si comió.}
  \upItem{upAccent}{Consideró instalar cámaras pero no encontró solución que la convenza.}}

\newcommand{\upEvidencia}{Construida a partir de encuesta (n\,=\,62; 82\,\% de cuidadores
con 45~años o más; 63\,\% con preocupación máxima; 50\,\% del adulto mayor vive solo;
84\,\% calificó el sistema con 4~o~5 sobre~5) y entrevista semiestructurada
con familiar cuidadora a distancia (2025).}
